function getSourceImageUrl(asset: any): string {
  return (
    asset.imageUrl ||
    asset.proxyUrl ||
    asset.url ||
    asset.publicUrl ||
    asset.downloadUrl ||
    ""
  );
}

function assetToSkinItem(asset: any): SkinItem {
  const imageUrl = getSourceImageUrl(asset);

  return {
    id: asset.id || asset.grfId,
    name: asset.name || asset.originalFilename || "Untitled source image",
    primaryImage: imageUrl,
    imageUrl,
    dimensions:
      asset.width && asset.height
        ? `${asset.width}x${asset.height}`
        : undefined,
    metadata: {
      grfId: asset.grfId || asset.id,
      mimeType: asset.mimeType,
      originalFilename: asset.originalFilename,
      channel: asset.channel,
      purpose: asset.purpose,
      sourceGrfId: asset.sourceGrfId,
      raw: asset,
    },
  };
}

Important:

- Keep the raw asset in metadata.raw so crop/delete/detail actions can access original GRF fields.
- Do not lose originalFilename.
- Do not lose mimeType.
- Do not lose grfId/id.

STEP 5 — FIX UPLOAD: ORIGINAL ONLY
============================================================

Upload behavior:

When a user uploads a file, create exactly ONE GRF record.

That record must be the original/source intake asset.

Use:

originalGrfParams(mimeType)

POST to:

/api/admin/graphics/save-grf

through adminFetch path:

/graphics/save-grf

Example:

async function saveSourceOriginal(file: File, imageDataBase64: string) {
  const mimeType = file.type || "image/jpeg";
  const grf = originalGrfParams(mimeType);

  return adminFetch("/graphics/save-grf", {
    method: "POST",
    body: JSON.stringify({
      ...grf,
      name: file.name,
      originalFilename: file.name,
      mimeType,
      imageUrl: `data:${mimeType};base64,${imageDataBase64}`,
    }),
  });
}

STEP 6 — FIX CROP: CREATE TWO GRF RECORDS
============================================================

Cropping is where the magic happens.

When crop completes, create TWO records automatically:

1. cropped derivative
2. background/promoted original

Use:

buildCropTransition(originalMimeType, "image/jpeg")

Example:

async function handleCropComplete(croppedDataUrl: string) {
  const originalAsset =
    selectedItem?.metadata?.raw || selectedAsset || assetToCrop;

  if (!originalAsset) {
    throw new Error("No original source asset selected for crop.");
  }

  const originalMimeType = originalAsset.mimeType || "image/jpeg";

  const transition = buildCropTransition(originalMimeType, "image/jpeg");

  const sourceGrfId =
    originalAsset.grfId ||
    originalAsset.id ||
    selectedItem?.metadata?.grfId ||
    selectedItem?.id;

  const originalImageUrl = getSourceImageUrl(originalAsset);

  await adminFetch("/graphics/save-grf", {
    method: "POST",
    body: JSON.stringify({
      ...transition.cropped,
      imageUrl: croppedDataUrl,
      name: `cropped_${originalAsset.name || originalAsset.originalFilename || sourceGrfId}`,
      mimeType: "image/jpeg",
      sourceGrfId,
      originalFilename: `cropped_${
        originalAsset.originalFilename || originalAsset.name || sourceGrfId
      }.jpg`,
    }),
  });

  await adminFetch("/graphics/save-grf", {
    method: "POST",
    body: JSON.stringify({
      ...transition.background,
      imageUrl: originalImageUrl,
      name: originalAsset.name || originalAsset.originalFilename || "background_source",
      mimeType: originalMimeType,
      sourceGrfId,
      originalFilename:
        originalAsset.originalFilename ||
        originalAsset.name ||
        `${sourceGrfId}.jpg`,
    }),
  });

  invalidateSourceCroppedBackgroundQueries();
}

Important:

- Cropped image uses croppedDataUrl.
