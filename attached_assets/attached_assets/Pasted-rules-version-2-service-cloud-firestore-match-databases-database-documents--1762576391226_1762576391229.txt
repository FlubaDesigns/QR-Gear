rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Helper functions
    function isSignedIn() {
      return request.auth != null;
    }
    
    function getUserData() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data;
    }
    
    function isAdmin() {
      return isSignedIn() && getUserData().role == 'admin';
    }
    
    // USERS - Open for testing
    match /users/{userId} {
      allow read: if true;
      allow create: if true;
      allow update: if request.auth.uid == userId || isAdmin();
      allow delete: if isAdmin();
    }
    
    // CHURCHES - Fully open for testing
    match /churches/{churchId} {
      allow read: if true;
      allow write: if isSignedIn();
    }
    
    // BUSINESS LISTINGS - Open reads, controlled writes
    match /business_listings/{businessId} {
      allow read: if true;
      allow create, update: if isSignedIn();
      allow delete: if isAdmin();
      
      match /commissions/{commissionId} {
        allow read: if isSignedIn();
        allow write: if isAdmin();
      }
      
      match /salesAnalytics/{analyticsId} {
        allow read: if isSignedIn();
        allow write: if isAdmin();
      }
    }
    
    // REVIEWS - Open for testing
    match /reviews/{reviewId} {
      allow read: if true;
      allow create, update: if isSignedIn();
      allow delete: if isAdmin();
    }
    
    // CATEGORIES - Open reads, admin writes
    match /categories/{categoryId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }
    
    // CATEGORY SUGGESTIONS
    match /category_suggestions/{suggestionId} {
      allow read: if isSignedIn();
      allow create: if isSignedIn();
      allow update, delete: if isAdmin();
    }
    
    // PAYMENTS - Keep secure even in testing
    match /payments/{paymentId} {
      allow read, write: if isAdmin();
    }
    
    // ACTIVITY LOG
    match /activity_log/{logId} {
      allow read: if isAdmin();
      allow create: if isSignedIn();
      allow update, delete: if isAdmin();
    }
    
    // FACTOIDS
    match /factoids/{factoidId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }
  }
}
